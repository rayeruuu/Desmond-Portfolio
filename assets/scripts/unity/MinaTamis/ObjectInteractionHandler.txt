    public bool TryHandleInteraction(RaycastHit hit, GameObject heldObject)
    {
        GameObject target = hit.collider.gameObject;
        string tag = target.tag;

        // Handle simple pickups with cached lookup
        if (SimplePickupTags.Contains(tag))
        {
            HandleSimplePickupOptimized(target, tag);
            return true;
        }

        // Handle tagged interactions with optimized lookup
        if (TryHandleTaggedInteractionOptimized(hit, target, tag, heldObject))
        {
            return true;
        }

        // Handle Mais Con Yelo containers (cache component lookup)
        MaisConYeloContainer maisContainer = target.GetComponent<MaisConYeloContainer>();
        if (maisContainer != null && maisContainer.currentState != MaisConYeloContainer.MaisState.Empty)
        {
            HandleMaisConYeloInteractions(target);
            return true;
        }

        return false;
    }
    
    private void HandleSimplePickupOptimized(GameObject target, string tag)
    {
        GameObject heldObject = manager.GetHeldObject();
        if (heldObject != null) return;
        
        // Special check for CookingPotLid - don't allow pickup when pot is boiling
        if (tag == "CookingPotLid")
        {
            // Check if this lid is attached to a boiling pot
            CookingPot parentPot = target.GetComponentInParent<CookingPot>();
            if (parentPot != null && parentPot.isBoiling)
            {
                manager.ShowInteraction("Can't remove lid while pot is boiling", true);
                return;
            }
        }
        
        Vector3 pickupOffset = PickupOffsets.TryGetValue(tag, out Vector3 offset) ? offset : PickupOffsets["Default"];
        Quaternion heldRotation = HeldRotations.TryGetValue(tag, out Quaternion rotation) ? rotation : default;
        
        manager.ShowPickupButton();
        // Removed duplicate item name display - now handled by ItemDescriptionDisplay
        if (Input.GetKeyDown(KeyCode.E))
        {
            // Special handling for CookingPot when picked up from stove
            if (tag == "CookingPot")
            {
                CookingPot pot = target.GetComponent<CookingPot>();
                if (pot != null && pot.onStove)
                {
                    pot.OnPickedUp(); // Call the pot's pickup method to handle stove cleanup
                }
            }
            // Note: CookingPan OnPickedUp() is handled in PickUpObject method to avoid duplicate calls
            
            PickUpObject(target, pickupOffset, heldRotation);
        }
    }
    
    private bool TryHandleTaggedInteractionOptimized(RaycastHit hit, GameObject target, string tag, GameObject heldObject)
    {
        // Use switch for better performance than dictionary lookup for execution
        switch (tag)
        {
            // Add this case for SimpleSink
            case "SimpleSink":
                return HandleSimpleSinkInteraction(target, heldObject);
                
            case "PlasticCup":
                HandleMaisConYeloInteractions(target);
                HandleHaloHaloInteractions(target);
                return true;
                
            case "CrushedIce":
                HandleMaisConYeloInteractions(target);
                HandleHaloHaloInteractions(target);
                return true;
                
            case "IceShredder":
                HandleIceShredderInteraction(target);
                return true;
                
            case "TuronWrapper":
                HandleTuronWrapperInteraction(hit);
                return true;
                
            case "KariokaPouch":
                HandleKariokaPouchPickup(target);
                return true;
                
            case "IngredientJar":
                HandleIngredientJarPickup(target);
                return true;
                
            case "BiloBox":
                HandleBiloBoxPickup(target);
                return true;
                
            case "BananaBox":
                HandleBananaBoxPickup(target);
                return true;
                
            case "Milk":
                HandleMilkPickup(target);
                return true;
                
            case "PalitawBox":
                HandlePalitawBoxInteraction(target);
                return true;
                
            case "PancakeBatter":
                HandlePancakeBatterPickup(hit);
                return true;
                
            case "JarHolder":
                HandleJarHolderInteraction(target);
                return true;
                
            case "Spatula":
                HandleSpatulaPickup(target);
                return true;
                
            case "IceCream":
                HandleIceCreamInteraction(target);
                return true;
                
            case "JarSnapPoint":
                HandleJarHolderInteraction(target.transform.parent.gameObject);
                return true;
                
            case "FrontWindow":
                HandleFrontWindowInteraction(hit);
                return true;
                
            case "Cabinet":
                HandleCabinetInteraction(hit);
                return true;
                
            case "RefDoor":
                HandleRefDoorInteraction(hit);
                return true;
                
            case "Door":
                HandleDoorInteraction(hit);
                return true;
                
            case "ButterStation":
                // Special case optimization
                if (heldObject != null && (IsCookedFood(heldObject) || heldObject.CompareTag("Plate")))
                {
                    return false; // Let FoodInteractionHandler handle it
                }
                return true;
                
            case "Stick":
                HandleStickInteraction(target);
                return true;
                
            case "SticksJar":
                HandleSticksJarInteraction(target);
                return true;
                
            case "SticksPackage":
                HandleSticksPackageInteraction(target);
                return true;
            case "OrderBox":
                // If we're holding an OrderBox and looking at a TrashCan, let FoodInteractionHandler handle it
                if (heldObject != null && heldObject.CompareTag("OrderBox") && target.CompareTag("TrashCan"))
                {
                    return false; // Let other handlers (FoodInteractionHandler) handle TrashCan interaction
                }
                HandleOrderBoxInteraction(target);
                return true;
            
            case "BananaTray":
                HandleBananaTrayInteraction(target);
                return true;
                
            case "KamoteTray":
                HandleKamoteTrayInteraction(target);
                return true;
                
            case "Radio":
                HandleRadioInteraction(target);
                return true;
                
            default:
                return false;
        }
    }


private void HandleOrderBoxInteraction(GameObject target)
{
    GameObject heldObject = manager.GetHeldObject();
    OrderBox orderBox = target.GetComponent<OrderBox>();
    
    if (orderBox == null) return;
    
    if (heldObject == null)
    {
        // Always allow pickup of the order box
        if (Input.GetKeyDown(KeyCode.E))
        {
            PickUpObject(target, PickupOffsets.ContainsKey("OrderBox") ? PickupOffsets["OrderBox"] : PickupOffsets["Default"]);
        }
    }
    else if (heldObject == target) // Player is holding this order box
    {

        
        if (orderBox.HasItems())
        {
            manager.ShowInteractButton();
            if (Input.GetKeyDown(KeyCode.E))
            {
                if (!orderBox.IsUIActive())
                {
                    orderBox.ShowUI();
                }
                else
                {
                    orderBox.HideUI();
                }
            }
        }
        else
        {

        }
    }
    else
    {

    }
}

    public bool TryHandleSnapInteraction(RaycastHit hit, GameObject heldObject)
    {
        if (heldObject == null) return false;

        // Use switch for better performance
        return heldObject.tag switch
        {
            "CookingPot" => HandleCookingPotSnap(hit, heldObject),
            "PancakeDispenser" => HandlePancakeDispenserSnap(hit, heldObject),
            "CookingPan" => HandleCookingPanSnap(hit, heldObject),
            "IngredientJar" => HandleIngredientJarSnap(hit, heldObject),
            _ => false
        };
    }

    public bool TryHandleCustomerInteraction(RaycastHit hit, GameObject heldObject)
    {

        HandleCustomerInteraction(customer, heldObject);

    private bool HandleCookingPotSnap(RaycastHit hit, GameObject heldObject)
    {
        CookingPot pot = heldObject.GetComponent<CookingPot>();
        if (pot == null) return false;

        // Check if there are available snap points (either new array system or old single system)
        bool hasAvailableSnapPoints = false;
        
        // Check new multiple snap points system
        if (pot.stoveParents != null && pot.stoveParents.Length > 0)
        {
            // Check if any snap point is available
            foreach (Transform snapPoint in pot.stoveParents)
            {
                if (snapPoint != null && !IsStoveSnapPointOccupied(snapPoint))
                {
                    hasAvailableSnapPoints = true;
                    break;
                }
            }
        }
        // Fallback to old single snap point system
        else if (pot.stoveParent != null)
        {
            hasAvailableSnapPoints = !IsStoveSnapPointOccupied(pot.stoveParent);
        }

        if (!hasAvailableSnapPoints) return false;

        // Optimized hit checking
        Transform hitTransform = hit.transform;
        bool isValidHit = hitTransform.CompareTag("Stove") || 
                         (pot.stoveParent != null && (hitTransform == pot.stoveParent || 
                         (hitTransform.parent != null && hitTransform.parent.CompareTag("Stove"))));

        if (isValidHit)
        {
            manager.ShowPlaceDownButton();
            if (cachedObjectPlacement != null) cachedObjectPlacement.showSnapPreview = true;

            if (Input.GetKeyDown(KeyCode.F) && !isPlacingPot)
                PlacePot(Vector3.zero, Quaternion.identity);

            return true;
        }

        return false;
    }

    private bool HandlePancakeDispenserSnap(RaycastHit hit, GameObject heldObject)
    {
        PancakeDispenser dispenser = heldObject.GetComponent<PancakeDispenser>();
        if (dispenser == null || dispenser.dispenserSnapPoint == null || !hit.transform.CompareTag("Mixer"))
            return false;

        manager.ShowPlaceDownButton();
        if (cachedObjectPlacement != null) cachedObjectPlacement.showSnapPreview = true;

        if (Input.GetKeyDown(KeyCode.F) && !isPlacingDispenser)
            PlaceDispenserAtSnap(dispenser);

        return true;
    }

    private bool HandleCookingPanSnap(RaycastHit hit, GameObject heldObject)
    {
        CookingPan pan = heldObject.GetComponent<CookingPan>();
        if (pan == null || !hit.transform.CompareTag("Stove"))
            return false;

        // Check if there are available snap points for this pan
        if (pan.stovePanSnapPoints == null || pan.stovePanSnapPoints.Length == 0)
        {
            // Fallback to old single snap point behavior
            if (pan.stovePanSnapPoint == null)
                return false;
        }

        manager.ShowPlaceDownButton();
        if (cachedObjectPlacement != null) cachedObjectPlacement.showSnapPreview = true;

        if (Input.GetKeyDown(KeyCode.F) && !isPlacingPan)
            PlacePanAtSnap(pan);

        return true;
    }

    private bool HandleIngredientJarSnap(RaycastHit hit, GameObject heldObject)
    {
        Transform hitTransform = hit.collider.transform;
        if (!hit.collider.CompareTag("JarSnapPoint") || hitTransform.childCount != 0)
            return false;

        manager.ShowPlaceDownButton();
        if (cachedObjectPlacement != null) cachedObjectPlacement.showSnapPreview = true;

        if (Input.GetKeyDown(KeyCode.F) && !isPlacingJar) // Prevent rapid placement
        {
            PlaceJarAtSnap(hitTransform, heldObject);
        }

        return true;
    }

    private void PlaceJarAtSnap(Transform snapPoint, GameObject jar)
    {
        if (isPlacingJar) return; // Double-check to prevent multiple placements
        
        isPlacingJar = true; // Set flag immediately
        
        // Store the original world scale BEFORE any operations
        Vector3 originalWorldScale = jar.transform.lossyScale;
        
        manager.ClearHeldObject();

        // First, detach from camera and reset parent
        jar.transform.SetParent(null);
        
        // Set position and rotation first
        jar.transform.position = snapPoint.position;
        jar.transform.rotation = snapPoint.rotation;
        
        // THEN parent to snap point
        jar.transform.SetParent(snapPoint);
        
        // Reset local position and rotation after parenting
        jar.transform.localPosition = Vector3.zero;
        jar.transform.localRotation = Quaternion.identity;
        
        // Calculate the correct local scale to maintain world scale
        Vector3 parentScale = snapPoint.lossyScale;
        Vector3 correctLocalScale = new Vector3(
            originalWorldScale.x / parentScale.x,
            originalWorldScale.y / parentScale.y,
            originalWorldScale.z / parentScale.z
        );
        jar.transform.localScale = correctLocalScale;
        
        SetLayerRecursively(jar, cachedDefaultLayer);

        // Optimized collider enabling
        EnableAllColliders(jar);

        // Play place down sound using AudioManager
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PLACEDOWN_SOUND, jar.transform.position);
        }

        Rigidbody rb = jar.GetComponent<Rigidbody>();
        if (rb != null)
        {
            rb.isKinematic = true;
            rb.useGravity = false;
        }

        if (cachedObjectPlacement != null) cachedObjectPlacement.showSnapPreview = false;
        
        // Reset flag after a short delay to prevent immediate re-placement
        manager.StartCoroutine(ResetPlacingFlag());
    }

    private System.Collections.IEnumerator ResetPlacingFlag()
    {
        yield return new WaitForSeconds(0.2f); // Small delay to prevent rapid placement
        isPlacingJar = false;
    }

    // Optimized helper method for enabling colliders
    private void EnableAllColliders(GameObject obj)
    {
        Collider[] colliders = obj.GetComponentsInChildren<Collider>();
        for (int i = 0; i < colliders.Length; i++)
        {
            colliders[i].enabled = true;
        }
    }

    private void HandleIceShredderInteraction(GameObject target)
    {
        GameObject heldObject = manager.GetHeldObject();
        if (heldObject != null && heldObject.CompareTag("IcePack"))
        {
            manager.ShowInteractButton();
            if (Input.GetKeyDown(KeyCode.E))
            {
                IceShredder shredder = target.GetComponent<IceShredder>();
                if (shredder != null)
                {
                    manager.StartCoroutine(shredder.ShredIceRoutine());
                    Object.Destroy(heldObject);
                    manager.ClearHeldObject();
                }
            }
        }
        else
        {
            manager.ShowInteraction("Need to hold an ice pack to use shredder");
        }
    }

    private void HandleFrontWindowInteraction(RaycastHit hit)
    {
        FrontWindow frontWindow = hit.collider.GetComponent<FrontWindow>();
        if (frontWindow == null) return;

        // Check if player can open the store (all previous objectives completed)
        if (!ObjectiveHelper.CanOpenStore())
        {
            string message = ObjectiveHelper.GetIncompleteObjectivesMessage();
            if (string.IsNullOrEmpty(message))
                message = "Complete all previous objectives before opening the store!";
            manager.ShowInteraction(message);
            return;
        }

        manager.ShowInteractButton();

        if (Input.GetKeyDown(KeyCode.E))
        {
            frontWindow.ToggleWindow();
        }
    }

    private void HandleJarHolderInteraction(GameObject target)
    {
        GameObject heldObject = manager.GetHeldObject();
        if (heldObject != null && heldObject.CompareTag("IngredientJar"))
        {
            ObjectPlacement objectPlacement = manager.GetObjectPlacement();
            if (objectPlacement != null) objectPlacement.showSnapPreview = true;

            Camera playerCamera = manager.GetPlayerCamera();
            Ray ray = new Ray(playerCamera.transform.position, playerCamera.transform.forward);
            RaycastHit hit;
            float rayDistance = 3f; // You might want to get this from manager

            if (Physics.Raycast(ray, out hit, rayDistance))
            {
                if (hit.collider.CompareTag("JarSnapPoint") && hit.collider.transform.childCount == 0)
                {
                    manager.ShowPlaceDownButton();

                    if (Input.GetKeyDown(KeyCode.F) && !isPlacingJar) // Prevent rapid placement here too
                    {
                        PlaceJarAtSnap(hit.collider.transform, heldObject);
                    }
                    return;
                }
            }
        }
        else
        {
            ObjectPlacement objectPlacement = manager.GetObjectPlacement();
            if (objectPlacement != null) objectPlacement.showSnapPreview = false;
        }
    }

    private void HandleTuronWrapperInteraction(RaycastHit hit)
    {
        TuronWrapper wrapper = hit.collider.gameObject.GetComponent<TuronWrapper>();
        if (wrapper == null) return;

        GameObject heldObject = manager.GetHeldObject();
        if (heldObject != null)
        {
            if (TryHandleWrapperBoxInteraction(wrapper)) return;
            if (TryHandleBananaBoxInteraction(wrapper, hit.collider.gameObject)) return;
            if (TryHandleIngredientJarInteraction(wrapper)) return;
        }
        else
        {
            HandleEmptyHandsTuronInteraction(wrapper, hit.collider.gameObject);
        }
    }

    private bool TryHandleWrapperBoxInteraction(TuronWrapper wrapper)
    {
        GameObject heldObject = manager.GetHeldObject();
        Wrapper wrapperBox = heldObject.GetComponent<Wrapper>();
        if (wrapperBox == null) return false;

        if (plate == null)

            return true;

            {
                if (!wrapperBox.TryPlaceWrapperOnPlate(plate))
                {
                }
            }
        }
        else
        {
        }
        return true;
    }

    private bool TryHandleIngredientJarInteraction(TuronWrapper wrapper)
    {
        GameObject heldObject = manager.GetHeldObject();
        IngredientJar jar = heldObject.GetComponent<IngredientJar>();
        if (jar == null || wrapper.isWrapped) return false;

        if (jar.jarType == IngredientJar.IngredientType.Sugar)
        {
            return HandleSugarJarInteraction(wrapper, jar);
        }

        if (jar.jarType == IngredientJar.IngredientType.Langka)
        {
            return HandleLangkaJarInteraction(wrapper, jar);
        }

        return false;
    }

    private bool HandleSugarJarInteraction(TuronWrapper wrapper, IngredientJar jar)

                                       });

    private void WrapTuron(TuronWrapper wrapper, GameObject wrapperObject)
    {
        Plate plate = wrapper.GetComponentInParent<Plate>();
        if (plate == null) return;

        Transform wrapperTransform = wrapperObject.transform;
        Vector3 originalPosition = wrapperTransform.position;
        Quaternion originalRotation = wrapperTransform.rotation;

        Wrapper wrapperBox = Object.FindFirstObjectByType<Wrapper>();
        if (!ValidateWrapperPrefab(wrapperBox)) return;

        CreateWrappedTuron(plate, wrapperObject, originalPosition, originalRotation, wrapperBox.wrappedFoodPrefab);
    }

    private bool ValidateWrapperPrefab(Wrapper wrapperBox)
    {
        if (wrapperBox == null || wrapperBox.wrappedFoodPrefab == null)
        {
            return false;
        }
        return true;
    }

    private void CreateWrappedTuron(Plate plate, GameObject wrapperObject, Vector3 position,
                                   Quaternion rotation, GameObject prefab)
    {
        plate.RemoveCurrentFood();
        Object.Destroy(wrapperObject);

        GameObject uncookedTuron = Object.Instantiate(prefab, position, rotation, plate.foodPlacementPoint);
        uncookedTuron.transform.localScale = new Vector3(0.15f, 0.15f, 0.15f);
        uncookedTuron.transform.localRotation = Quaternion.Euler(0f, 180f, 0f);

        SetupCookableFood(uncookedTuron);
        plate.SetCurrentFood(uncookedTuron);
    }

    }

    private void HandleCustomerInteraction(CustomerNPC customer, GameObject heldObject)
    {
        // First check if holding finished food directly (like food on stick)
        if (heldObject.CompareTag("FinishedFood"))
        {
            CookableFood cookableFood = heldObject.GetComponent<CookableFood>();
            if (cookableFood != null && cookableFood.foodState == CookableFood.FoodState.FinishedFood)
            {
                // Special case: Karioka and Kamotecue on stick must be placed on plate first
                if (cookableFood.foodType == CookableFood.FoodType.Karioka || 
                    cookableFood.foodType == CookableFood.FoodType.Kamotecue)
                {
                    string foodName = cookableFood.foodType == CookableFood.FoodType.Karioka ? "Karioka" : "Kamotecue";
                    manager.ShowInteraction($"Put the {foodName} on a plate first", true);
                    return;
                }
                
                // For other finished foods, allow direct serving
                manager.ShowInteractButton();
                if (Input.GetKeyDown(KeyCode.E))
                {
                    if (customer.CheckOrder(heldObject))
                    {
                        // Complete serve objectives based on food type
                        CookableFood heldFood = heldObject.GetComponent<CookableFood>();
                        if (heldFood != null && heldFood.foodState == CookableFood.FoodState.FinishedFood)
                        {
                            // For days 1-6, customer serving is handled by GameManager.CustomerServed()
                            // which tracks the "Serve to customers 10x" objective
                            // Individual food serving objectives are no longer used for learning days
                            
                            // Note: GameManager.CustomerServed() will be called when the customer is actually served
                        }
                        
                        Object.Destroy(heldObject);
                        manager.ClearHeldObject();
                    }
                    else
                    {
                        manager.ShowInteraction("Customer doesn't want this food", true);
                    }
                }
                return;
            }
        }

        // Handle plate interactions
        Plate plate = heldObject.GetComponent<Plate>();
        if (plate != null)
        {
            if (plate.HasFood() || plate.HasCookedPalitaw())
            {
                manager.ShowInteractButton();
                if (Input.GetKeyDown(KeyCode.E))
                {
                    GameObject foodToCheck = plate.GetCurrentFood();
                    if (foodToCheck != null)
                    {
                        if (customer.CheckOrder(foodToCheck))
                        {
                            CookableFood plateFood = foodToCheck.GetComponent<CookableFood>();
                            if (plateFood != null && plateFood.foodState == CookableFood.FoodState.FinishedFood)
                            {
                                // For days 1-6, customer serving is handled by GameManager.CustomerServed()
                                // which tracks the "Serve to customers 10x" objective
                                // Individual food serving objectives are no longer used for learning days
                                
                                // Note: GameManager.CustomerServed() will be called when the customer is actually served
                                // This section is just for objective completion of individual food preparation steps
                            }
                            Object.Destroy(foodToCheck);
                            plate.RemoveCurrentFood();
                            Object.Destroy(heldObject);
                            manager.ClearHeldObject();
                        }
                        else
                        {
                            manager.ShowInteraction("Customer doesn't want this food", true);
                        }
                    }
                    else
                    {
                        GameObject palitaw = plate.GetCookedPalitaw();
                        if (palitaw != null && customer.CheckOrder(palitaw))
                        {
                            Object.Destroy(palitaw);
                            plate.RemoveCurrentFood();
                            Object.Destroy(heldObject);
                            manager.ClearHeldObject();
                        }
                        else
                        {
                            manager.ShowInteraction("Customer doesn't want this food", true);
                        }
                    }
                }
            }
            return;
        }

        // Handle other cookable foods (but not finished foods on stick - those are handled above)
        CookableFood otherCookableFood = heldObject.GetComponent<CookableFood>();
        if (otherCookableFood != null && !heldObject.CompareTag("FinishedFood"))
        {
            manager.ShowInteractButton();
            manager.ShowInteraction("Give food to customer");
            if (Input.GetKeyDown(KeyCode.E))
            {
                if (customer.CheckOrder(heldObject))
                {
                    Object.Destroy(heldObject);
                    manager.ClearHeldObject();
                }
                else
                {
                    manager.ShowInteraction("Customer doesn't want this food", true);
                }
            }
        }
    }

    // Placement methods
    // Public method to reset all placement flags (called when picking up objects or cancelling interactions)
    public void ResetPlacementFlags()

        isPlacingPot = false;

    public void PlaceHeldObject(Vector3 position, Quaternion rotation)
    {
        GameObject heldObject = manager.GetHeldObject();
        if (heldObject == null) return;

        // Reset placement flags when placing any object
        ResetPlacementFlags();

        // Special snap placement for cooking pots and pans
        ObjectPlacement objectPlacement = manager.GetObjectPlacement();
        Transform previewSnapPoint = objectPlacement?.GetPreviewSnapPoint();
        
        Debug.Log($"PlaceHeldObject: Held object = {heldObject.tag}, Preview snap point = {(previewSnapPoint != null ? previewSnapPoint.name : "null")}");
        
        // Handle cooking pot snap placement
        if (heldObject.CompareTag("CookingPot") && previewSnapPoint != null)
        {
            CookingPot pot = heldObject.GetComponent<CookingPot>();
            if (pot != null)
            {
                Debug.Log($"PlaceHeldObject: Placing cooking pot at snap point {previewSnapPoint.name} at position {previewSnapPoint.position}");
                
                // Use the snap placement method with the specific snap point
                bool success = pot.PlaceAtStoveSnapPoint(manager.GetDefaultLayer(), previewSnapPoint);
                
                if (success)
                {
                    // Clear held object and hide preview
                    manager.ClearHeldObject();
                    if (objectPlacement != null)
                        objectPlacement.DestroyPreview();
                    
                    // Play place down sound
                    if (AudioManager.Instance != null)
                    {
                        AudioManager.Instance.PlaySFX3D(DEFAULT_PLACEDOWN_SOUND, previewSnapPoint.position);
                    }
                    
                    return;
                }
                else
                {
                    Debug.LogWarning("Failed to place cooking pot at snap point, falling back to regular placement");
                }
            }
        }
        
        // Handle cooking pan snap placement
        if (heldObject.CompareTag("CookingPan") && previewSnapPoint != null)
        {
            CookingPan pan = heldObject.GetComponent<CookingPan>();
            if (pan != null)
            {
                Debug.Log($"PlaceHeldObject: Placing cooking pan at snap point {previewSnapPoint.name} at position {previewSnapPoint.position}");
                
                // Use the snap placement method with the specific snap point
                bool success = pan.PlaceAtStoveSnapPoint(manager.GetDefaultLayer(), previewSnapPoint);
                
                if (success)
                {
                    // Clear held object and hide preview
                    manager.ClearHeldObject();
                    if (objectPlacement != null)
                        objectPlacement.DestroyPreview();
                    
                    // Play place down sound
                    if (AudioManager.Instance != null)
                    {
                        AudioManager.Instance.PlaySFX3D(DEFAULT_PLACEDOWN_SOUND, previewSnapPoint.position);
                    }
                    
                    return;
                }
                else
                {
                    Debug.LogWarning("Failed to place cooking pan at snap point, falling back to regular placement");
                }
            }
        }

        // Special handling for OrderBox
        if (heldObject.CompareTag("OrderBox"))
        {
            OrderBox orderBox = heldObject.GetComponent<OrderBox>();
            if (orderBox != null)
            {
                orderBox.OnPlacedDown();
            }
        }

        // Special handling for preserving component states
        PancakeDispenser heldDispenser = null;
        int savedPancakeCount = 0;
        CookingOilContainer heldOilContainer = null;

        if (heldObject != null)
        {
            heldDispenser = heldObject.GetComponent<PancakeDispenser>();
            if (heldDispenser != null)
                savedPancakeCount = heldDispenser.GetRemainingCount();
                
            // Save oil container state
            heldOilContainer = heldObject.GetComponent<CookingOilContainer>();
        }

        heldObject.transform.SetParent(null);
        GameObject placedObject = Object.Instantiate(heldObject, heldObject.transform.position, heldObject.transform.rotation);

        Plate newPlate = placedObject.GetComponent<Plate>();
        if (newPlate != null)
        {
            Transform foodPoint = newPlate.foodPlacementPoint;
            if (foodPoint != null && foodPoint.childCount > 0)
            {
                GameObject newFood = foodPoint.GetChild(0).gameObject;
                newPlate.SetCurrentFood(newFood);
            }
            else
            {
                newPlate.SetCurrentFood(null);
            }
        }
        
        // Restore oil container state if needed
        if (heldOilContainer != null)
        {
            CookingOilContainer newOilContainer = placedObject.GetComponent<CookingOilContainer>();
            if (newOilContainer != null)
            {
                newOilContainer.TransferStateFrom(heldOilContainer);
            }
        }

        Object.Destroy(heldObject);
        manager.ClearHeldObject();

        // Play place down sound using AudioManager
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PLACEDOWN_SOUND, position);
        }

        SetLayerRecursively(placedObject, manager.GetDefaultLayer());

        Collider[] colliders = placedObject.GetComponentsInChildren<Collider>();
        foreach (var col in colliders)
        {
            col.enabled = false;
        }

        manager.StartCoroutine(MovePlacedObject(placedObject, position, rotation, colliders, heldDispenser, savedPancakeCount));

        if (objectPlacement != null)
            objectPlacement.DestroyPreview();
    }

    private void PlaceCookingPotWithWater(GameObject pot, Vector3 position, Quaternion rotation)
    {
        // Don't destroy and recreate - just move the existing pot
        pot.transform.SetParent(null);
        manager.ClearHeldObject();
        
        SetLayerRecursively(pot, manager.GetDefaultLayer());
        
        // Play place down sound using AudioManager
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PLACEDOWN_SOUND, position);
        }
        
        Collider[] colliders = pot.GetComponentsInChildren<Collider>();
        foreach (var col in colliders)
        {
            col.enabled = false;
        }
        
        manager.StartCoroutine(MoveCookingPot(pot, position, rotation, colliders));
        
        ObjectPlacement objectPlacement = manager.GetObjectPlacement();
        if (objectPlacement != null)
            objectPlacement.DestroyPreview();
    }

    private System.Collections.IEnumerator MoveCookingPot(GameObject pot, Vector3 targetPos, Quaternion targetRot, Collider[] colliders)
    {
        float duration = 0.2f;
        float elapsed = 0f;

        Vector3 startPos = pot.transform.position;
        Quaternion startRot = pot.transform.rotation;

        Rigidbody rb = pot.GetComponent<Rigidbody>();
        if (rb == null)
            rb = pot.AddComponent<Rigidbody>();

        rb.isKinematic = true;

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            float t = Mathf.Clamp01(elapsed / duration);
            pot.transform.position = Vector3.Lerp(startPos, targetPos, t);
            pot.transform.rotation = Quaternion.Slerp(startRot, targetRot, t);
            yield return null;
        }

        pot.transform.position = targetPos;
        pot.transform.rotation = targetRot;

        yield return new WaitForSeconds(0.1f);

        foreach (var col in colliders)
            col.enabled = true;

        if (rb != null)
        {
            rb.isKinematic = true;
            rb.useGravity = false;
        }

        // Update the EnvironmentObjectID to store the new position as the initial state
        // This prevents objects from floating when the next day starts
        EnvironmentObjectID envObjectID = pot.GetComponent<EnvironmentObjectID>();
        if (envObjectID != null)
        {
            envObjectID.UpdateInitialState();
        }
        else
        {
            // If the object doesn't have an EnvironmentObjectID, add one and set its initial state
            envObjectID = pot.AddComponent<EnvironmentObjectID>();
            envObjectID.UpdateInitialState();
        }
    }

    private void PlacePot(Vector3 position, Quaternion rotation)
    {
        if (isPlacingPot) return; // Prevent multiple placements
        
        GameObject heldObject = manager.GetHeldObject();
        if (heldObject == null || !heldObject.CompareTag("CookingPot")) return;

        CookingPot potScript = heldObject.GetComponent<CookingPot>();
        if (potScript == null) return;

        isPlacingPot = true; // Set flag immediately

        GameObject pot = heldObject;
        manager.ClearHeldObject();

        // Get the preview snap point and use it for placement
        ObjectPlacement objectPlacement = manager.GetObjectPlacement();
        Transform previewSnapPoint = objectPlacement?.GetPreviewSnapPoint();
        
        bool success = false;
        if (previewSnapPoint != null)
        {
            Debug.Log($"ObjectInteractionHandler.PlacePot: Using preview snap point {previewSnapPoint.name}");
            success = potScript.PlaceAtStoveSnapPoint(manager.GetDefaultLayer(), previewSnapPoint);
        }
        else
        {
            Debug.LogWarning("ObjectInteractionHandler.PlacePot: No preview snap point, using fallback");
            success = potScript.PlaceAtStoveSnapPoint(manager.GetDefaultLayer());
        }

        // Play place down sound using AudioManager
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PLACEDOWN_SOUND, position);
        }

        if (success && objectPlacement != null)
            objectPlacement.DestroyPreview();
        else if (!success)
        {
            manager.SetHeldObject(pot);
            Debug.LogWarning("No available snap points for cooking pot placement");
        }
        
        isPlacingPot = false; // Reset flag
    }

    public void PickUpObject(GameObject obj, Vector3 positionOffset, Quaternion customRotation = default)
{
    GameObject heldObject = manager.GetHeldObject();
    if (heldObject != null) return;

    // Check if picking up a plate to complete objective
    if (obj.CompareTag("Plate"))
    {
        ObjectiveHelper.CompleteActiveObjective(ObjectiveHelper.Day.MinaTamisDebut, ObjectiveHelper.MinaTamisDebutObjectives.GetPlate);
    }

    // Reset all placement flags when picking up any object
    ResetPlacementFlags();

    manager.SetHeldObject(obj);
    SetLayerRecursively(obj, manager.GetPlayerLayer());

    Collider[] colliders = obj.GetComponentsInChildren<Collider>();
    foreach (var col in colliders)
        col.enabled = false;

    Rigidbody rb = obj.GetComponent<Rigidbody>();
    if (rb != null)
    {
        rb.isKinematic = true;
        rb.useGravity = false;
    }

    Camera playerCamera = manager.GetPlayerCamera();
    obj.transform.SetParent(playerCamera.transform);

    // Special handling for OrderBox
    if (obj.CompareTag("OrderBox"))
    {
        OrderBox orderBox = obj.GetComponent<OrderBox>();
        if (orderBox != null)
        {
            orderBox.OnPickedUp();
        }
        
        // Play pickup sound using AudioManager
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D("PickUp", obj.transform.position);
        }
    }
    // Special handling for CookingPot - reset stove flags
    else if (obj.CompareTag("CookingPot"))
    {
        CookingPot pot = obj.GetComponent<CookingPot>();
        if (pot != null)
        {
            pot.OnPickedUp();
        }
        
        // Play general pickup sound
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PICKUP_SOUND, obj.transform.position);
        }
    }
    // Special handling for CookingPan - reset stove flags
    else if (obj.CompareTag("CookingPan"))
    {
        CookingPan pan = obj.GetComponent<CookingPan>();
        if (pan != null)
        {
            pan.OnPickedUp();
        }
        
        // Play general pickup sound
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PICKUP_SOUND, obj.transform.position);
        }
    }
    else if (obj.CompareTag("IcePack"))
    {
        // Play ice pickup sound using AudioManager
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D("IcePickup", obj.transform.position);
        }
    }
    else
    {
        // Play general pickup sound for all other objects
        if (AudioManager.Instance != null)
        {
            AudioManager.Instance.PlaySFX3D(DEFAULT_PICKUP_SOUND, obj.transform.position);
        }
    }

    manager.StartCoroutine(PositionHeldObject(obj, positionOffset, customRotation));
    manager.ClearInteraction();
}

    private System.Collections.IEnumerator PositionHeldObject(GameObject obj, Vector3 positionOffset, Quaternion targetRotation = default)
    {
        if (obj == null) yield break;

        Vector3 startPos = obj.transform.localPosition;
        Quaternion startRot = obj.transform.localRotation;

        // Fixed: Default rotation is now Y=-180 for ALL objects (except spatula and stick which use custom rotations)
        if (targetRotation == default)
            targetRotation = Quaternion.Euler(0, -180, 0);

        float duration = 0.2f;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            if (obj == null) yield break;

            elapsed += Time.deltaTime;
            float t = Mathf.Clamp01(elapsed / duration);

            obj.transform.localPosition = Vector3.Lerp(startPos, positionOffset, t);
            obj.transform.localRotation = Quaternion.Slerp(startRot, targetRotation, t);

            yield return null;
        }

        if (obj != null)
        {
            obj.transform.localPosition = positionOffset;
            obj.transform.localRotation = targetRotation;

            Collider[] colliders = obj.GetComponentsInChildren<Collider>();
            foreach (var col in colliders)
                col.enabled = true;
        }
    }

    public void SetLayerRecursively(GameObject obj, int layer)
    {
        if (obj == null) return;
        obj.layer = layer;
        foreach (Transform child in obj.transform)
        {
            SetLayerRecursively(child.gameObject, layer);
        }
    }

private void HandleStickInteraction(GameObject target)
{
    GameObject heldObject = manager.GetHeldObject();
    
    if (heldObject == null)
    {
        // Check if this stick belongs to a jar
        SticksJar parentJar = target.GetComponentInParent<SticksJar>();
        if (parentJar != null)
        {
            manager.ShowPickupButton();
            if (Input.GetKeyDown(KeyCode.E))
            {
                // Store the original world scale BEFORE any operations
                Vector3 originalWorldScale = target.transform.lossyScale;
                
                // Simple: Just duplicate and disable the original
                GameObject duplicateStick = Object.Instantiate(target);
                duplicateStick.name = "HeldStick";
                duplicateStick.tag = "Stick";
                duplicateStick.transform.SetParent(null);
                duplicateStick.SetActive(true);
                
                // Restore the world scale after instantiation
                duplicateStick.transform.localScale = originalWorldScale;
                
                // Disable the original stick in the jar
                target.SetActive(false);
                parentJar.StickTaken(target);
                
                // Refresh jar state to ensure accuracy
                parentJar.RefreshJarState();
                
                // Stick uses Y=-90 rotation (different from default Y=-180)
                PickUpObject(duplicateStick, new Vector3(0.3f, -0.3f, 0.5f), Quaternion.Euler(0, -90, 0));
            }
        }
        else
        {
            // For loose sticks
            manager.ShowPickupButton();
            if (Input.GetKeyDown(KeyCode.E))
            {
                // Stick uses Y=-90 rotation (different from default Y=-180)
                PickUpObject(target, new Vector3(0.3f, -0.3f, 0.5f), Quaternion.Euler(0, -90, 0));
            }
        }
    }
    else if (heldObject.CompareTag("Stick"))
    {
    }
    else
    {
        manager.ShowInteraction("Can't take stick while holding something else");
    }
}

private void HandleSticksJarInteraction(GameObject target)
{
    GameObject heldObject = manager.GetHeldObject();
    SticksJar sticksJar = target.GetComponent<SticksJar>();
    
    if (sticksJar == null)
    {

        return;
    }
    
    if (heldObject != null && heldObject.CompareTag("Stick"))
    {
        if (sticksJar.CanAcceptStick())
        {
            manager.ShowInteractButton();
            if (Input.GetKeyDown(KeyCode.E))
            {
                if (sticksJar.ReturnStick())
                {
                    Object.Destroy(heldObject);
                    manager.ClearHeldObject();
                }
                else
                {
                }
            }
        }
        else
        {
        }
    }
    else if (heldObject != null && heldObject.CompareTag("SticksPackage"))
    {
        SticksPackage package = heldObject.GetComponent<SticksPackage>();
        if (package != null)
        {
            if (sticksJar.IsEmpty())
            {
                manager.ShowInteractButton();
                // Package refill count now shown by ItemDescriptionDisplay - show action only
                manager.ShowInteraction("Refill jar");
                if (Input.GetKeyDown(KeyCode.E))
                {
                    Debug.Log("Attempting to refill jar...");
                    sticksJar.DebugJarStatus(); // Debug jar state before refill
                    
                    if (package.RefillJar(sticksJar))
                    {
                        ObjectiveHelper.CompleteActiveObjective(ObjectiveHelper.Day.ThreeSweetTreats, ObjectiveHelper.ThreeSweetTreatsObjectives.RefillStickHolder);
                        Debug.Log("Jar refill successful!");
                        sticksJar.RefreshJarState(); // Refresh state after refill
                        sticksJar.DebugJarStatus(); // Debug jar state after refill
                        
                        // Check if package was destroyed (no refills left)
                        if (package == null || package.refillsRemaining <= 0)
                        {
                            manager.ClearHeldObject();
                        }
                    }
                    else
                    {
                        Debug.LogWarning("Jar refill failed!");
                        sticksJar.DebugJarStatus(); // Debug why it failed
                    }
                }
            }
            else
            {
                int remainingSticks = sticksJar.GetRemainingSticks();
                manager.ShowInteraction($"Jar must be empty to refill (has {remainingSticks} sticks)");
                
                // Debug: Allow force empty for testing by holding F key
                if (Input.GetKey(KeyCode.F))
                {
                    manager.ShowInteraction("Hold F to force empty jar (DEBUG)");
                    if (Input.GetKeyDown(KeyCode.E))
                    {
                        sticksJar.ForceEmpty();
                        Debug.Log("DEBUG: Jar forcibly emptied!");
                    }
                }
            }
        }
    }
    else if (heldObject == null)
    {
        // Allow picking up sticks from jar when hands are empty
        if (sticksJar.HasSticksAvailable())
        {
            manager.ShowPickupButton();
            if (Input.GetKeyDown(KeyCode.E))
            {
                GameObject stickToTake = sticksJar.GetAvailableStick();
                if (stickToTake != null)
                {
                    // Store the original world scale BEFORE any operations
                    Vector3 originalWorldScale = stickToTake.transform.lossyScale;
                    
                    // Create a duplicate stick for the player to hold
                    GameObject duplicateStick = Object.Instantiate(stickToTake);
                    duplicateStick.name = "HeldStick";
                    duplicateStick.tag = "Stick";
                    duplicateStick.transform.SetParent(null);
                    duplicateStick.SetActive(true);
                    
                    // Restore the world scale after instantiation
                    duplicateStick.transform.localScale = originalWorldScale;
                    
                    // Disable the original stick in the jar
                    stickToTake.SetActive(false);
                    sticksJar.StickTaken(stickToTake);
                    
                    // Refresh jar state to ensure accuracy
                    sticksJar.RefreshJarState();
                    
                    // Stick uses Y=-90 rotation (different from default Y=-180)
                    PickUpObject(duplicateStick, new Vector3(0.3f, -0.3f, 0.5f), Quaternion.Euler(0, -90, 0));
                }
            }
        }
        // Stick count now shown by ItemDescriptionDisplay
    }
    else
    {
        manager.ShowInteraction("Can only return sticks or use sticks package with jar");
    }
}

private bool HandleSimpleSinkInteraction(GameObject target, GameObject heldObject)
{
    SimpleSink sink = target.GetComponent<SimpleSink>();
    if (sink == null) return false;
    
    // If no object is held, don't handle this interaction
    if (heldObject == null) return false;
    
    // Check if the sink can actually be activated for this object
    if (sink.CanActivate(heldObject))
    {
        // Show the cleaning message
        string cleaningMessage = sink.GetCleaningMessage(heldObject);
        if (!string.IsNullOrEmpty(cleaningMessage))
        {
            manager.ShowInteractButton();
            manager.ShowInteraction(cleaningMessage.ToLower());
            
            if (Input.GetKeyDown(KeyCode.E))
            {
                sink.StartCleaning(heldObject);
            }
        }
        return true; // We handled this interaction - prevents F key placement
    }
    
    // IMPORTANT: If sink can't be activated for this object, return false
    return false;
}

// Check if a stove snap point is already occupied by another cooking pot or pan
private bool IsStoveSnapPointOccupied(Transform snapPoint)
{
    if (snapPoint == null) return true;

    // Find all cooking pots and pans in the scene
    CookingPot[] allPots = UnityEngine.Object.FindObjectsByType<CookingPot>(FindObjectsSortMode.None);
    CookingPan[] allPans = UnityEngine.Object.FindObjectsByType<CookingPan>(FindObjectsSortMode.None);

    float snapThreshold = 0.5f; // Distance threshold to consider something "at" the snap point

    // Check if any pot is at this snap point
    foreach (CookingPot pot in allPots)
    {
        if (pot.isSnappedToStove && Vector3.Distance(pot.transform.position, snapPoint.position) < snapThreshold)
        {
            return true;
        }
    }

    // Check if any pan is at this snap point
    foreach (CookingPan pan in allPans)
    {
        if (pan.isSnappedToStove && Vector3.Distance(pan.transform.position, snapPoint.position) < snapThreshold)
        {
            return true;
        }
    }

    return false;
}
